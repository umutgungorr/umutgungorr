name: System & Codebase Forensics
on: [push, pull_request]

jobs:
  diagnostic-gate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: umutgungorr/porttree@v1     # Pre-Flight Socket & Port Collision Guard
